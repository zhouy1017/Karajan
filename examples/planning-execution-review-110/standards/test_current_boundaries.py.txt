import pytest
from tests.orchestration.test_planning_execution import planning_case, begin, configured

pytest_plugins = ['tests.runs.test_planning']

def prepared(tmp_path, configured):
    service, run, intent, authorities = planning_case(tmp_path, configured)
    execution = begin(service, run, intent, authorities)
    authorities.activate()
    return service, run, execution, authorities


def test_source_change_after_capture_before_claim_is_rejected(configured, tmp_path, monkeypatch):
    service, run, execution, authorities = prepared(tmp_path, configured)
    claim = service._claim_submission
    monkeypatch.setattr(service, '_claim_submission', lambda execution_id, principal, key: service.get(execution_id, principal=principal))
    result = service.submit(execution['id'], principal='owner', command_key='submit')
    assert result['state'] == 'output_captured'
    authorities.output_source = 'c' * 64
    monkeypatch.setattr(service, '_claim_submission', claim)
    service.submit(execution['id'], principal='owner', command_key='submit')
    assert service.planner.get(run['id'], principal='owner')['plans'] == []


def test_cancel_before_claim_is_final_and_prevents_plan(configured, tmp_path, monkeypatch):
    service, run, execution, authorities = prepared(tmp_path, configured)
    claim = service._claim_submission
    def cancel_then_claim(*args):
        result = service.cancel(execution['id'], principal='owner', command_key='cancel')
        assert result['state'] == 'cancelled'
        return claim(*args)
    monkeypatch.setattr(service, '_claim_submission', cancel_then_claim)
    result = service.submit(execution['id'], principal='owner', command_key='submit')
    assert result['state'] == 'cancelled'
    assert service.planner.get(run['id'], principal='owner')['plans'] == []


def test_post_claim_crash_preserves_unknown_without_second_bridge(configured, tmp_path, monkeypatch):
    service, run, execution, authorities = prepared(tmp_path, configured)
    calls=[]
    def crash(*args, **kwargs):
        calls.append(1)
        raise SystemExit('crash')
    monkeypatch.setattr(service.planner, '_submit_planning_execution_plan', crash)
    with pytest.raises(SystemExit):
        service.submit(execution['id'], principal='owner', command_key='submit')
    result = service.submit(execution['id'], principal='owner', command_key='submit')
    assert result['state'] == 'submission_unknown'
    assert calls == [1]
    assert service.planner.get(run['id'], principal='owner')['plans'] == []


def test_lost_submission_reply_recovers_with_output_authority_unavailable(configured, tmp_path, monkeypatch):
    service, run, execution, authorities = prepared(tmp_path, configured)
    original=service.planner._submit_planning_execution_plan
    calls=[]
    def lose(*args, **kwargs):
        calls.append(1)
        original(*args, **kwargs)
        raise RuntimeError('lost reply')
    monkeypatch.setattr(service.planner, '_submit_planning_execution_plan', lose)
    with pytest.raises(RuntimeError):
        service.submit(execution['id'], principal='owner', command_key='submit')
    service.outputs=None
    service.admissions=None
    result=service.submit(execution['id'], principal='owner', command_key='submit')
    assert result['state']=='submitted'
    assert calls==[1]
    assert service.planner.get(run['id'], principal='owner')['plans']==[result['submission']]
