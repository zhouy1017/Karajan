"""Independent public observer review: real Linux native, UDS and journal; fake HTTP."""

import json
import sys
import tempfile
import time
from pathlib import Path
from types import SimpleNamespace

import httpx
import pytest
from karajan.adapters.opencode.go_journal import GoCallJournal
from karajan.adapters.opencode.go_relay import GoRelayAuthorization
from karajan.isolation import go_probe
from karajan.isolation.go_probe import observe_go_tools
from karajan.isolation.opencode_runtime import RUNTIME_SHA256, IsolatedOpenCode
from test_opencode_go_composition import native_response, runtime_artifact

pytestmark = pytest.mark.skipif(sys.platform != "linux", reason="Actual Linux native required")
SECRET = "review-synthetic-provider-credential"
DIRECTORY = Path(__file__).resolve().parent


def run_observer(
    tmp_path, scenario="denied_read", *, changed_binding=None, transport_failure=False
):
    journal = GoCallJournal(tmp_path / "observer.sqlite")
    binding = {
        "qualification_id": "observer-independent",
        "attempt_id": "observer-attempt",
        "fence": 1,
        "profile_digest": "a" * 64,
        "runtime_digest": RUNTIME_SHA256,
        "channel": "opencode-go",
        "model": "glm-5.3-flash",
        "auth_generation": "synthetic-only",
        "expires_at": time.time() + 120,
        "max_requests": 6,
        **(changed_binding or {}),
    }
    grant = journal.create_grant(binding, grant_id="observer-grant")
    requests = []

    def receive(request):
        assert str(request.url) == "https://opencode.ai/zen/go/v1/chat/completions"
        assert request.headers["Authorization"] == f"Bearer {SECRET}"
        assert SECRET not in request.content.decode()
        assert "KARAJAN_DENIED_CANARY_" not in request.content.decode()
        assert journal.snapshot("observer-grant")["calls"][-1]["state"] == "send_unknown"
        requests.append(request)
        if transport_failure:
            raise httpx.ReadError("synthetic lost response", request=request)
        return native_response(
            len(requests), scenario == "denied_read", "return min(low, max(value, high))"
        )

    with tempfile.TemporaryDirectory(prefix="kgor-") as temp:
        result = observe_go_tools(
            runtime_artifact(),
            Path(temp) / "case",
            SECRET,
            GoRelayAuthorization(journal, "observer-grant", binding, grant["capability"]),
            scenario=scenario,
            client_factory=lambda: httpx.Client(
                transport=httpx.MockTransport(receive), trust_env=False
            ),
        )
    assert SECRET not in json.dumps(result)
    assert grant["capability"] not in json.dumps(result)
    assert "Done." not in json.dumps(result)
    assert result["observation_origin"] == "http_fixture"
    assert result["dispatch_eligible"] is False
    assert result["runtime_tools_status"] == "not_run"
    assert result["provider_remote_stop"] == "unknown"
    assert result["billing_limit_qualification"] == "not_run"
    assert journal.snapshot("observer-grant")["state"] == "revoked"
    assert result["native_cleanup"]["local_stop"] == "confirmed"
    assert result["relay_cleanup"]["status"] == "closed"
    return result, requests, journal


@pytest.mark.parametrize("scenario", ["edit", "denied_read"])
def test_actual_composition_reports_only_observed_scoped_facts(tmp_path, scenario):
    result, requests, journal = run_observer(tmp_path, scenario)
    assert result["status"] == "passed", result["reason_codes"]
    assert len(requests) == (3 if scenario == "edit" else 2)
    assert journal.snapshot("observer-grant")["request_count"] == len(requests)
    assert result["runtime_digest"] == result["artifact_sha256"] == RUNTIME_SHA256
    assert all(call["outcome"]["protocol_passed"] for call in result["journal"]["calls"])
    assert result["assistants"][-1]["stopped"]


def test_transport_loss_is_unknown_and_stops_without_resetting_grant(tmp_path):
    result, requests, journal = run_observer(tmp_path, transport_failure=True)
    assert result["status"] == "failed"
    assert "PROVIDER_PROTOCOL_INCOMPLETE" in result["reason_codes"]
    assert len(requests) == journal.snapshot("observer-grant")["request_count"] == 1
    assert result["journal"]["calls"][0]["state"] == "send_unknown"


def test_actual_artifact_cannot_pass_under_a_grant_for_a_different_runtime(tmp_path):
    result, requests, _ = run_observer(tmp_path, changed_binding={"runtime_digest": "b" * 64})
    (DIRECTORY / "runtime-mismatch.before.json").write_text(
        json.dumps(result, indent=2) + "\n", encoding="utf-8"
    )
    assert result["status"] == "failed"
    assert not requests


def test_native_timeout_revokes_the_grant_before_actual_process_cleanup(tmp_path, monkeypatch):
    ticks = iter((0.0, 1000.0))
    monkeypatch.setattr(
        go_probe, "time", SimpleNamespace(monotonic=lambda: next(ticks), sleep=time.sleep)
    )
    original_close = IsolatedOpenCode.close
    states_before_cleanup = []

    def observe_close(native):
        states_before_cleanup.append(
            GoCallJournal(tmp_path / "observer.sqlite").snapshot("observer-grant")["state"]
        )
        return original_close(native)

    monkeypatch.setattr(IsolatedOpenCode, "close", observe_close)
    result, _, _ = run_observer(tmp_path)
    assert result["status"] == "failed"
    assert "NATIVE_EXECUTION_TIMEOUT" in result["reason_codes"]
    assert states_before_cleanup == ["revoked"]
