@pytest.mark.parametrize("lost_index", [1, 2, 3])
def test_lost_create_reply_revokes_only_preallocated_owned_grants(
    synthetic_suite, monkeypatch, lost_index
):
    suite, journal, start = synthetic_suite
    original, count = journal.create_grant, 0

    def lost(binding, *, grant_id):
        nonlocal count
        result = original(binding, grant_id=grant_id)
        count += 1
        if count == lost_index:
            raise ConnectionError(credential().reveal())
        return result

    monkeypatch.setattr(journal, "create_grant", lost)
    result = suite.observe(start, credential())
    assert result["status"] == "failed"
    assert [row["status"] for row in result["scenarios"]] == ["not_run"] * 3
    assert [row["state"] for row in result["grant_cleanup"]] == (
        ["revoked"] * lost_index + ["not_created"] * (3 - lost_index)
    )
    assert credential().reveal() not in str(result)
    for row in start["scenarios"][:lost_index]:
        assert (
            GoCallJournal(journal.path, clock=lambda: 101.0).snapshot(row["grant_id"])[
                "request_count"
            ]
            == 0
        )

def test_clock_becomes_unavailable_after_effect_retains_cleanup(synthetic_suite, monkeypatch):
    suite, journal, start = synthetic_suite

    def observed(runtime, directory, secret, authorization, *, scenario, **kwargs):
        report = synthetic_observation(suite, authorization, scenario)

        def missing():

        journal.snapshot(row["grant_id"])["state"] == "revoked" for row in start["scenarios"]
    )


@pytest.mark.parametrize(
    "change",
    [
        "credential",
        "expiry",
        "future",
        "fence",
        "source",
        "model",
        "context",
        "schema",
        "duplicate_attempt",
        "extra",
    ],
)
def test_start_mismatch_is_zero_grants_and_zero_observations(synthetic_suite, monkeypatch, change):
    from dataclasses import replace

    from karajan.adapters.opencode.go_journal import GoJournalError

    suite, journal, start = synthetic_suite
    secret = credential()
    if change == "credential":
        secret = replace(secret, generation="new-generation")
    elif change == "expiry":
        start["expires_at"] = 701.0
        for row in start["scenarios"]:
            row["grant_binding"]["expires_at"] = 701.0
    elif change == "future":
        suite.clock = lambda: 99.0
    elif change == "source":
        start["source"]["observation_origin"] = "official_go"
    elif change == "duplicate_attempt":
        start["scenarios"][1]["attempt_id"] = start["scenarios"][0]["attempt_id"]
        start["scenarios"][1]["grant_binding"]["attempt_id"] = start["scenarios"][0]["attempt_id"]
    elif change == "extra":
        start["scenarios"][0]["grant_binding"]["suite_ref"] = start["suite_ref"]
    elif change == "context":
        start["scenarios"][0]["grant_binding"]["context"] = {
            **start["scenarios"][0]["grant_binding"]["context"],
            "fixed_margin": 1,
        }
    else:
        key, value = {
            "fence": ("fence", True),
            "model": ("model", "other-model"),
            "schema": ("schema_version", "karajan.go-qualification-grant.v2"),
        }[change]
        start["scenarios"][0]["grant_binding"][key] = value
    called = []
    install_observer(monkeypatch, lambda *args, **kwargs: called.append(True))
    with pytest.raises(ValueError, match="FIXED_GO_REVIEWER_START_BINDING_MISMATCH"):
        suite.observe(start, secret)
    assert called == []
    with pytest.raises(GoJournalError, match="GRANT_NOT_FOUND"):
        journal.snapshot(start["scenarios"][0]["grant_id"])

def test_grant_send_unknown_cannot_be_overwritten_by_a_passed_report(synthetic_suite, monkeypatch):
    suite, journal, start = synthetic_suite
    original = journal.complete_call

    def no_commit(grant_id, call_id, **kwargs):
        # Lost completion before commit leaves actual begin intent unknown.
        return {"outcome": kwargs["outcome"]}

    monkeypatch.setattr(journal, "complete_call", no_commit)
    install_observer(
        monkeypatch,
        lambda runtime, directory, secret, authorization, *, scenario, **kwargs: (
            synthetic_observation(suite, authorization, scenario)
        ),
    )
    result = suite.observe(start, credential())
    monkeypatch.setattr(journal, "complete_call", original)
    snapshot = GoCallJournal(journal.path, clock=lambda: 101.0).snapshot(
        start["scenarios"][0]["grant_id"]
    )
    assert result["status"] == "failed"
    assert snapshot["state"] == "revoked"
    assert [call["state"] for call in snapshot["calls"]] == ["send_unknown", "send_unknown"]

@pytest.mark.parametrize(
    "scenario,change",
    [
        ("defect_review", "line"),
        ("defect_review", "blocking"),
        ("defect_review", "acceptance"),
        ("denied_read", "passed"),
        ("denied_read", "not_denied"),
    ],
)
def test_known_scenario_semantics_are_reparsed_not_trusted_from_passed(
    synthetic_suite, monkeypatch, scenario, change
):
    from karajan.candidates.review_output import parse_review_output

    suite, journal, start = synthetic_suite

    def observed(runtime, directory, secret, authorization, *, scenario, **kwargs):
        report = synthetic_observation(suite, authorization, scenario)
        if scenario != target:
            return report
        if change == "not_denied":
            report["tools"][0]["permission_denied"] = False
            return report
        body = json.loads(report["native_final"]["text"])
        if change == "line":
            body["findings"][0]["line"] = 1
        elif change == "blocking":
            body["findings"][0]["blocking"] = False
        elif change == "acceptance":
            body["findings"][0]["acceptance_ref"] = "acceptance:foreign"
        else:
            body["verdict"] = "pass"
        text = json.dumps(body)
        report["native_final"]["text"] = text
        report["native_final"]["text_sha256"] = hashlib.sha256(text.encode()).hexdigest()
        # All changes except the foreign reference also pass the real parser:
        # the Suite must enforce fixed-case expectations beyond JSON validity.
        if change != "acceptance":
            report["parsed_review"] = parse_review_output(
                text,
                allowed_files=frozenset({"acceptance.md", "src/range.py"}),
                allowed_acceptance_refs=frozenset({"acceptance:clamp-v1"}),
            ).model_dump(mode="json")
        return report

    target = scenario
    install_observer(monkeypatch, observed)
    result = suite.observe(start, credential())
    assert result["status"] == "failed"
    row = next(row for row in result["scenarios"] if row["scenario"] == scenario)
    assert row["status"] == "failed"
    assert row["reason_codes"] != ["SCENARIO_OBSERVATION_INCOMPLETE"]
