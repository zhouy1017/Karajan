def deploy(approved, directory, image):
    """Explicit test deployment copies existing logical stores via SQLite backup."""
    state, control, host_root, work = (
        directory / name for name in ("state", "control", "host", "work")
    )
    for path in (state, control, host_root, work):
        path.mkdir(mode=0o700)
    admission = approved.admissions
    for name, source in {
        "projects.sqlite": admission.routing.planner.projects.database,
        "runs.sqlite": admission.routing.planner.database,
        "capacity.sqlite": admission.routing.capacity.path,
        "task-admissions.sqlite": admission.database,
    }.items():
        with (
            closing(sqlite3.connect(source)) as original,
            closing(sqlite3.connect(state / name)) as target,
        ):
            original.backup(target)
    RunnerHost(host_root)
    approved.candidates.directory.chmod(0o700)
    settings = CheckSettings(
        control,
        state,
        approved.candidates.directory,
        host_root,
        work,
        Path(sys.executable),
        tuple(admission.routing.planner.projects.allowed_roots),
        (CheckEnvironmentSource("stdlib", 1, image.directory),),
    )
    write_check_bootstrap(settings)
    return settings


@pytest.mark.parametrize("defective", [False, True])
def test_fixed_host_child_runs_every_approved_check_and_reopens_without_reexecution(
    projected, tmp_path, defective
):
    image = PythonCheckEnvironment.provision(tmp_path / "image")
    environment = {
        "id": "stdlib",
        "revision": 1,
        "runtime_kind": "python312-stdlib",
        "platform": "linux_x64",
        "source_sha256": image.source()["environment_sha256"],
        "filesystem": "candidate_copy",
        "network": "none",
        "env": {},
        "max_log_bytes": 16384,
    }
    reference = {"id": "stdlib", "revision": 1}
    approved = approved_check_candidate(
        projected,
        tmp_path / "approved",
        environment=environment,
        checks=[
            {
                "id": "behavior",
                "revision": 1,
                "environment_ref": reference,
                "timeout_seconds": 10,
                "argv": [
                    "python3.12",
                    "-I",
                    "-c",
                    "from pathlib import Path; "
                    "assert Path('src/report.py').read_bytes() == b\"print('collected')\\n\"; "
                    "print('behavior-ok')",
                ],
            },
            {
                "id": "syntax",
                "revision": 1,
                "environment_ref": reference,
                "timeout_seconds": 10,
                "argv": ["python3.12", "-I", "-m", "py_compile", "src/report.py"],
            },
        ],
        worker_output=b"print('defective')\n" if defective else b"print('collected')\n",
    )
    settings = deploy(approved, tmp_path, image)
    run_id, operation_id = approved.args
    service = open_check_services(
        settings, run_id=run_id, operation_id=operation_id, principal="owner", for_execution=True
    )
    deadline = time.monotonic() + 120
    try:
        while time.monotonic() < deadline:
            service.advance(run_id, operation_id, principal="owner")
            service.reconcile(run_id, operation_id, principal="owner")
            validation = service.get(run_id, operation_id, principal="owner")
            rows = validation["checks"]["runs"] if validation else []
            if len(rows) == 2 and all(row.get("evidence") is not None for row in rows):
                break
            time.sleep(0.1)
        else:
            status = service.get(run_id, operation_id, principal="owner")
            (tmp_path / "public-validation.json").write_text(json.dumps(status, indent=2))
            pytest.fail("Fixed Check entry did not record both required checks")
        subject = validation["subject"]["candidate"]
        candidate = service.candidates.get(subject["id"])
        gate = service.candidates.gate(
            candidate["id"],
            current={
                key: candidate[key]
                for key in ("repository_identity", "base_sha", "input_sha256", "policy_sha256")
            },
        )
        checks = {
            row["input"]["check_id"]: row for row in gate["evidence"] if row["kind"] == "check"
        }
        assert set(checks) == {"behavior", "syntax"}
        assert checks["behavior"]["status"] == ("failed" if defective else "passed")
        assert checks["syntax"]["status"] == "passed"
        assert all(row["input"]["provenance"] == "trusted_observation" for row in checks.values())
        assert gate["local_gate_passed"] is gate["delivery_eligible"] is False
        assert "REVIEW_EVIDENCE_MISSING" in gate["reasons"]
        assert (approved.repository / "src/report.py").read_bytes() == b"print('approved task')\n"
        original_ids = {name: row["id"] for name, row in checks.items()}
        history = open_check_services(
            settings, run_id=run_id, operation_id=operation_id, principal="owner"
        )
        for _ in range(2):
            history.reconcile(run_id, operation_id, principal="owner")
        after = history.candidates.gate(
            candidate["id"],
            current={
                key: candidate[key]
                for key in ("repository_identity", "base_sha", "input_sha256", "policy_sha256")
            },
        )
        assert {
            row["input"]["check_id"]: row["id"]
            for row in after["evidence"]
            if row["kind"] == "check"
        } == original_ids
        assert len(after["evidence"]) == 2
    finally:
        service.cancel(run_id, operation_id, principal="owner")
