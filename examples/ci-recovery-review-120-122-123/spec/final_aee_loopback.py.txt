"""Independent exact-Git-source loopback review; preserves all earlier scripts/results.

Only fixture artifact metadata, budget, endpoint, and trusted ephemeral local CA
are substituted. Default transport/handlers/HTTPResponse remain the candidate's.
"""
import hashlib
import http.server
import json
from pathlib import Path
import ssl
import subprocess
import threading
import time
import types
import uuid

ROOT = Path(__file__).resolve().parents[2]
PRIVATE = Path(__file__).resolve().parent
SHA = 'aee3ca8269c622cb96833ea1b8f2477fb3e773e9'
source = subprocess.check_output(['git', 'show', SHA + ':.github/scripts/provision_go_tokenizer.py'], cwd=ROOT)
cert_dir = PRIVATE / ('loopback-cert-' + uuid.uuid4().hex)
cert_dir.mkdir()
cert, key = cert_dir / 'certificate.pem', cert_dir / 'key.pem'
subprocess.run(['C:/Program Files/Git/usr/bin/openssl.exe', 'req', '-x509', '-newkey', 'rsa:2048', '-nodes', '-keyout', str(key), '-out', str(cert), '-days', '1', '-subj', '/CN=localhost', '-addext', 'subjectAltName=IP:127.0.0.1'], check=True, capture_output=True)
results = []
for case in ['slow_headers', 'slow_body', 'slow_chunk_size', 'slow_trailer', 'later_read', 'fixed_hold', 'chunk_tcp_split', 'eof', 'redirect_budget', 'redirect_tls_handshake', 'retry_budget', 'bad_digest']:
    module = types.ModuleType('review_' + case)
    exec(compile(source, SHA + '/provision_go_tokenizer.py', 'exec'), module.__dict__)
    budget = 0.8 if case == 'later_read' else (2.0 if case in ('chunk_tcp_split', 'eof') else 0.3)
    body = b'x' * (65537 if case == 'later_read' else 16)
    stop = threading.Event()
    requests = []
    class Handler(http.server.BaseHTTPRequestHandler):
        def log_message(self, *args): pass
        def do_GET(self):
            requests.append(self.path)
            def write(raw):
                self.wfile.write(raw); self.wfile.flush()
            def trickle(raw):
                for byte in raw:
                    if stop.wait(0.07): return
                    write(bytes([byte]))
            try:
                if case in ('redirect_budget', 'redirect_tls_handshake') and self.path == '/':
                    stop.wait(0.22)
                    write(('HTTP/1.1 302 Found\r\nContent-Length: 0\r\nLocation: https://127.0.0.1:' + str(self.server.server_port) + '/final\r\n\r\n').encode())
                    return
                if case == 'redirect_budget': stop.wait(0.22)
                if case == 'retry_budget':
                    stop.wait(0.12)
                    write(b'HTTP/1.1 503 Service Unavailable\r\nContent-Length: 0\r\n\r\n')
                    return
                if case == 'slow_headers':
                    write(b'HTTP/1.1 200 OK\r\nContent-Length: 16\r\nX-Slow: ')
                    trickle(b'abcdefghijklmnop')
                    write(b'\r\n\r\n' + body)
                    return
                chunked = case in ('slow_chunk_size', 'slow_trailer', 'chunk_tcp_split')
                header = b'Transfer-Encoding: chunked\r\n' if chunked else (b'' if case == 'eof' else b'Content-Length: ' + str(len(body)).encode() + b'\r\n')
                write(b'HTTP/1.1 200 OK\r\n' + header + b'Connection: close\r\n\r\n')
                if case == 'slow_body': trickle(body)
                elif case == 'later_read':
                    stop.wait(0.5); write(body[:-1]); stop.wait(0.6); write(body[-1:])
                elif case == 'slow_chunk_size':
                    trickle(b'00000000000010\r\n'); write(body + b'\r\n0\r\n\r\n')
                elif case == 'slow_trailer':
                    write(b'10\r\n' + body + b'\r\n0\r\nX-Trailer: ')
                    trickle(b'abcdefghijklmnop'); write(b'\r\n\r\n')
                elif case == 'chunk_tcp_split':
                    for part in [b'1', b'0;extension=value\r', b'\n' + body[:2], body[2:8], body[8:] + b'\r', b'\n0\r\nX-Trailer: yes\r', b'\n\r', b'\n']:
                        write(part); stop.wait(0.02)
                else:
                    write(b'y' * len(body) if case == 'bad_digest' else body)
                    if case == 'fixed_hold': stop.wait(0.8)
            except (BrokenPipeError, ConnectionResetError, ssl.SSLError): pass
    server_tls = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
    server_tls.load_cert_chain(str(cert), str(key))
    class TLSHTTPServer(http.server.ThreadingHTTPServer):
        accepted = 0
        def get_request(self):
            sock, address = self.socket.accept()
            self.accepted += 1
            if case == 'redirect_tls_handshake' and self.accepted == 2:
                stop.wait(0.22)
            return server_tls.wrap_socket(sock, server_side=True), address
    server = TLSHTTPServer(('127.0.0.1', 0), Handler)
    client_tls = ssl.create_default_context(cafile=str(cert))
    thread = threading.Thread(target=server.serve_forever, daemon=True); thread.start()
    original_build = module.build_opener
    def local_build(*handlers):
        for handler in handlers:
            if isinstance(handler, module._DeadlineHTTPSHandler): handler._context = client_tls
        return original_build(*handlers)
    module.build_opener = local_build
    original_open = module._open
    module._open = lambda _, *, timeout=30.0: original_open('https://127.0.0.1:' + str(server.server_port) + '/', timeout=timeout)
    module.ARTIFACTS = {'fixture.bin': (len(body), hashlib.sha256(body).hexdigest())}
    module.DOWNLOAD_BUDGET_SECONDS = budget
    target = PRIVATE / ('stream-run-' + uuid.uuid4().hex); target.mkdir()
    old = b'old file must survive'
    (target / 'fixture.bin').write_bytes(old)
    started = time.monotonic(); reason = None
    try: module.provision(target)
    except Exception as error: reason = type(error).__name__ + ':' + str(error)
    elapsed = time.monotonic() - started
    success = case in ('fixed_hold', 'chunk_tcp_split', 'eof')
    data = (target / 'fixture.bin').read_bytes()
    passed = (reason is None and data == body) if success else (reason is not None and data == old and elapsed < budget + 0.09)
    if case == 'bad_digest': passed = passed and len(requests) == 1
    temps = len(list(target.glob('*.tmp')))
    results.append({'case': case, 'budget_seconds': budget, 'elapsed_seconds': round(elapsed, 3), 'reason': reason, 'requests': requests, 'old_preserved': data == old, 'valid_published': data == body, 'temps': temps, 'passed': passed and temps == 0})
    stop.set(); server.shutdown(); server.server_close(); thread.join()
print(json.dumps({'candidate': SHA, 'source_sha256': hashlib.sha256(source).hexdigest(), 'script_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(), 'external_traffic': False, 'TLS_verification': 'enabled; trusts generated local CA only', 'results': results}, indent=2))
