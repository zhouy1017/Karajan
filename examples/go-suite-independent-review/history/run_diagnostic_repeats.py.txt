import os,subprocess,sys
from pathlib import Path
root=Path('.cache/go-suite-independent-review')
for index in range(1,4):
    result=subprocess.run([sys.executable,'-m','pytest',str(root/'test_go_suite_review.py'),'-q','-k','caller_start','-p','no:cacheprovider',f'--basetemp={root}/repeat-{index}-tmp',f'--junitxml={root}/repeat-{index}.xml'],env=os.environ.copy())
    if result.returncode:
        raise SystemExit(result.returncode)
