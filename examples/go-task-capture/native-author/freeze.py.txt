import hashlib
import json
import xml.etree.ElementTree as ET
from pathlib import Path

sources = [
    'backend/karajan/isolation/opencode_runtime.py',
    'backend/karajan/isolation/_opencode_capture.py',
    'tests/isolation/test_opencode_projection_capture.py',
]
context = [
    'backend/karajan/candidates/store.py',
    'tests/candidates/test_projected_capture.py',
    'tests/candidates/test_baseline_materialization.py',
    'tests/isolation/test_opencode_go_composition.py',
    'backend/karajan/isolation/_opencode_projection.py',
    'backend/karajan/isolation/_opencode_namespace.py',
    'backend/karajan/isolation/_opencode_inner.py',
]
names = ['tracer-red', 'tracer-green', 'readonly-red', 'native-edit', 'final', 'windows', 'regression']
evidence = ['.cache/projection-capture/README.md'] + [
    f'.cache/projection-capture/{name}.xml' for name in names
]

def hashes(paths):
    return {p: hashlib.sha256(Path(p).read_bytes()).hexdigest() for p in paths}

def results():
    value = {}
    for name in names:
        suite = ET.parse(f'.cache/projection-capture/{name}.xml').getroot().find('testsuite')
        value[name] = {k: suite.get(k) for k in ['tests', 'errors', 'failures', 'skipped', 'time']}
    return value

freeze = {
    'base': '1534825',
    'kind': 'author_evidence',
    'source_sha256': hashes(sources),
    'consumption_context_sha256': hashes(context),
    'evidence_sha256': hashes(evidence),
    'junit': results(),
    'checks': {
        'ruff_check': 'All checks passed',
        'ruff_format_check': '3 files already formatted',
        'mypy_windows': 'Success: no issues found in 2 source files',
        'mypy_linux': 'Success: no issues found in 2 source files',
    },
    'limits': [
        'Synthetic HTTP upstream; no actual provider or credential',
        'Actual fixed native ELF, namespace and read/edit tools exercised',
        'Synthetic Candidate Actor/Writer binding; no approved Run capture claim',
        'Unknown OS wait/poll receipt test is fault injection after actual pidfd kill',
        'Local confirmed stop never attests remote inference stop',
        'No project Git mutation or CI failure repair',
    ],
}
Path('.cache/projection-capture/freeze.json').write_text(
    json.dumps(freeze, indent=2, ensure_ascii=False) + '\n', encoding='utf-8'
)
print(json.dumps({'sources': freeze['source_sha256'], 'junit': freeze['junit']}, indent=2))
