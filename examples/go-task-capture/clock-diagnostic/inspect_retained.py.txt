"""Read retained synthetic evidence; export only safe summary fields, never DB bytes."""

import hashlib
import json
import sqlite3
from pathlib import Path

from karajan.isolation.go_probe import go_runtime_source, source_digest

root = Path('/tmp/pytest-of-chow/pytest-43/test_fixed_suite_cross_checks_1')
artifact = Path('/mnt/c/Users/Chooo/Playground/Karajan/.cache/go-linux-runtime/package/bin/opencode')
dbpath = root / 'journal.sqlite'
source = go_runtime_source(artifact)
db = sqlite3.connect(dbpath.as_uri() + '?mode=ro', uri=True)
db.row_factory = sqlite3.Row
grants = []
for row in db.execute('SELECT id,binding,created_at,revoked_at FROM go_grants ORDER BY created_at'):
    bound = json.loads(row['binding'])
    count = db.execute('SELECT COUNT(*) FROM go_calls WHERE grant_id=?', (row['id'],)).fetchone()[0]
    grants.append({
        'grant_id': row['id'],
        'created_at': row['created_at'],
        'revoked_at': row['revoked_at'],
        'expires_at': bound['expires_at'],
        'request_count': count,
        'runtime_source_matches_current': bound['runtime_digest'] == source_digest(source),
    })
db.close()
started = grants[0]['expires_at'] - 420
report = {
    'mode': 'read_only_original_inspection_and_deterministic_public_boundary_replay',
    'runtime_artifact': str(artifact),
    'original': {
        'retained_path': str(root),
        'original_present': root.is_dir(),
        'journal_sha256': hashlib.sha256(dbpath.read_bytes()).hexdigest(),
        'grants': grants,
        'derived_started_at': started,
        'derivation': 'unchanged execution_start sets expires_at = started_at + 420',
        'minimum_observed_clock_rollback_seconds': started - grants[0]['created_at'],
        'suite_directories': {
            p.name: sorted(c.name for c in p.iterdir()) for p in root.glob('s*') if p.is_dir()
        },
        'failure_case_seconds': 4.136,
        'actual_statuses': ['not_run', 'not_run'],
        'expected_statuses': ['failed', 'not_run'],
        'native_launched': False,
        'observer_reached': False,
    },
    'conclusion': 'Retained facts show a wall-clock rollback before the scenario start guard. The conservative rejection is expected; no production time-rule change is required.',
    'limits': [
        'The exact second suite-clock sample was not persisted; it is not invented here.',
        'The deterministic replay uses a timestamp between recorded grant creation and the derived start, not a claimed recording of the original sample.',
        'No native process or upstream provider is used by the replay.',
        'Original Journal bytes, capabilities, credential values and raw native files are not exported.',
    ],
}
Path(__file__).with_name('report.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps({
    'original_present': report['original']['original_present'],
    'derived_started_at': started,
    'minimum_observed_clock_rollback_seconds': report['original']['minimum_observed_clock_rollback_seconds'],
    'all_sources_match': all(g['runtime_source_matches_current'] for g in grants),
    'counts': [g['request_count'] for g in grants],
    'directory_contents': report['original']['suite_directories'],
}))
